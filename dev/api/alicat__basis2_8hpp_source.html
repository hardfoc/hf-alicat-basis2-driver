147 DriverResult<InstantaneousData> ReadInstantaneous(uint8_t flow_decimals = 3) noexcept {
148 std::array<uint16_t, 10> regs{};
149 const auto rc = ReadHoldingRegisters(reg::SelectedGas, regs.data(),
150 static_cast<uint8_t>(regs.size()));
151 if (rc != DriverError::None) return DriverResult<InstantaneousData>::failure(rc);
152
153 InstantaneousData out{};
154 out.gas = static_cast<Gas>(regs[0] & 0xFFU);
155
156 const uint16_t status = regs[1];
157 out.mass_flow_overrange = (status & 0x0001U) != 0;
158 out.temperature_overrange = (status & 0x0002U) != 0;
159 out.totalizer_overrange = (status & 0x0004U) != 0;
160 out.valve_held = (status & 0x0008U) != 0;
161 out.valve_thermal_management = (status & 0x0010U) != 0;
162
163 out.raw_temp_2102 = static_cast<int16_t>(regs[2]);
164 out.temperature_c = static_cast<float>(out.raw_temp_2102) * 0.01f;
165
166 out.raw_flow_2103 = static_cast<int16_t>(regs[3]);
167 out.flow = ScaleFlow(out.raw_flow_2103, flow_decimals);
168
169 out.raw_total_volume_2104_2105 =
170 (static_cast<uint32_t>(regs[4]) << 16) | regs[5];
171 out.total_volume = ScaleFlow(static_cast<int32_t>(out.raw_total_volume_2104_2105),
172 flow_decimals);
173
174 out.raw_setpoint_2053_2054 = static_cast<int16_t>(regs[6]);
175 out.setpoint = ScaleFlow(out.raw_setpoint_2053_2054, flow_decimals);
176
177 out.valve_drive_pct = static_cast<float>(static_cast<int16_t>(regs[7])) * 0.01f;
178
179 out.raw_batch_remaining_2108_2109 =
180 (static_cast<uint32_t>(regs[8]) << 16) | regs[9];
181 out.batch_remaining = ScaleFlow(static_cast<int32_t>(out.raw_batch_remaining_2108_2109),
182 flow_decimals);
183
184 return DriverResult<InstantaneousData>::success(out);
185 }

201 DriverResult<InstrumentIdentity> ReadIdentity() noexcept {
202 InstrumentIdentity id{};
203
204 // Firmware version
205 std::array<uint16_t, 1> fw{};
206 if (auto rc = ReadHoldingRegisters(reg::FirmwareVersion, fw.data(), 1);
207 rc != DriverError::None) {
208 return DriverResult<InstrumentIdentity>::failure(rc);
209 }
210 id.firmware_version_raw = fw[0];
211 id.fw_major = static_cast<uint8_t>((fw[0] >> 8) & 0xFFU);
212 id.fw_minor = static_cast<uint8_t>((fw[0] >> 4) & 0x0FU);
213 id.fw_patch = static_cast<uint8_t>(fw[0] & 0x0FU);
214
215 // Serial number
216 std::array<uint16_t, reg::SerialNumberCount> sn{};
217 if (auto rc = ReadHoldingRegisters(reg::SerialNumberStart, sn.data(),
218 reg::SerialNumberCount);
219 rc != DriverError::None) {
220 return DriverResult<InstrumentIdentity>::failure(rc);
221 }
222 for (std::size_t i = 0; i < reg::SerialNumberCount; ++i) {
223 id.serial_number[i * 2] = static_cast<char>((sn[i] >> 8) & 0xFFU);
224 id.serial_number[i * 2 + 1] = static_cast<char>(sn[i] & 0xFFU);
225 }
226 id.serial_number[12] = '\0';
227
228 // Address + unit id + flow units
229 std::array<uint16_t, 5> tail{}; // 45..49
230 if (auto rc = ReadHoldingRegisters(reg::ModbusAddress, tail.data(),
231 static_cast<uint8_t>(tail.size()));
232 rc != DriverError::None) {
233 return DriverResult<InstrumentIdentity>::failure(rc);
234 }
235 id.modbus_address = static_cast<uint8_t>(tail[0] & 0xFFU);
236 id.ascii_unit_id = static_cast<char>(tail[1] & 0xFFU);
237 id.full_scale_flow_raw =
238 (static_cast<uint32_t>(tail[2]) << 16) | tail[3];
239 id.flow_units = static_cast<FlowUnits>(tail[4] & 0xFFU);
240 return DriverResult<InstrumentIdentity>::success(id);
241 }

461 DriverResult<MeasurementData> ReadMeasurement(uint8_t flow_decimals = 3) noexcept {
462 constexpr uint16_t kFirst = reg::MeasurementSampleCount; // 4201
463 constexpr uint16_t kLast = reg::MeasurementPrevAvgFlow + reg::kPrevAvgFlowCount - 1; // 4215
464 std::array<uint16_t, kLast - kFirst + 1> regs{};
465 const auto rc = ReadHoldingRegisters(kFirst, regs.data(), static_cast<uint8_t>(regs.size()));
466 if (rc != DriverError::None)
467 return DriverResult<MeasurementData>::failure(rc);
468 const auto at = [&regs](uint16_t r) { return regs[r - kFirst]; };
469
470 MeasurementData out{};
471 out.num_samples_requested = at(reg::MeasurementSampleCount);
472 out.num_samples_collected = at(reg::MeasurementSamplesAveraged);
473 out.min_temperature_raw = static_cast<int16_t>(at(reg::MeasurementMinTemp));
474 out.max_temperature_raw = static_cast<int16_t>(at(reg::MeasurementMaxTemp));
475 out.min_flow_raw = static_cast<int16_t>(at(reg::MeasurementMinFlow));
476 out.max_flow_raw = static_cast<int16_t>(at(reg::MeasurementMaxFlow));
477 out.avg_temperature_raw = static_cast<int16_t>(at(reg::MeasurementAvgTemp));
478 out.avg_flow_raw = static_cast<int16_t>(at(reg::MeasurementAvgFlow));

578 DriverResult<uint8_t> DiscoverPresentAddresses(uint8_t* present_bitmap,
579 std::size_t bitmap_bytes,
580 uint16_t probe_timeout_ms = 30,
581 uint8_t first_addr = 1,
582 uint8_t last_addr = kMaxModbusAddress) noexcept {
583 if (!present_bitmap || bitmap_bytes < 32) {
584 return DriverResult<uint8_t>::failure(DriverError::BufferTooSmall);
585 }
586 std::memset(present_bitmap, 0, bitmap_bytes);
587
588 const uint8_t saved_addr = address_;
589 const uint16_t saved_timeout = timeout_ms_;
590 timeout_ms_ = probe_timeout_ms;
591 quiet_ = true; // silent addresses are expected: do not log each timeout
592
593 uint8_t found = 0;
594 for (uint16_t a = first_addr; a <= last_addr; ++a) {
595 address_ = static_cast<uint8_t>(a);
596 std::array<uint16_t, 1> dummy{};
597 if (ReadHoldingRegisters(reg::FirmwareVersion, dummy.data(), 1) ==
598 DriverError::None) {
599 present_bitmap[a / 8] |= static_cast<uint8_t>(1U << (a % 8));
600 ++found;
601 }
602 }
603
604 address_ = saved_addr;
605 timeout_ms_ = saved_timeout;
606 quiet_ = false;
607 return DriverResult<uint8_t>::success(found);
608 }

620 DriverError ReadHoldingRegisters(uint16_t reg_addr, uint16_t* out, uint8_t count) noexcept {
621 if (!out || count == 0 || count > 125) return DriverError::InvalidParameter;
622 std::array<uint8_t, 8> tx{};
623 const std::size_t tx_len = modbus::BuildReadHolding(address_, reg_addr, count,
624 tx.data(), tx.size());
625 if (tx_len == 0) return DriverError::BufferTooSmall;
626
627 // Expected response: addr(1) + fn(1) + bytecount(1) + payload(2*count) + crc(2)
628 const std::size_t need = 5 + 2U * count;
629 std::array<uint8_t, 5 + 2 * 32> rx{}; // up to 32 regs per call
630 if (need > rx.size()) return DriverError::BufferTooSmall;
631
632 DriverError rc = SendRequest(tx.data(), tx_len);
633 if (rc != DriverError::None) {
634 return LogFailure(rc, reg::fn::ReadHolding, reg_addr);
635 }
636 std::size_t got = 0;
637 rc = ReceiveReply(reg::fn::ReadHolding, rx.data(), need, got);
638 if (rc == DriverError::None) {
639 rc = modbus::ParseReadHolding(rx.data(), got, address_, count, out);
640 }
641 return LogFailure(rc, reg::fn::ReadHolding, reg_addr);
642 }

650 DriverResult<void> WriteSingleRegister(uint16_t reg_addr, uint16_t value) noexcept {
651 std::array<uint8_t, 8> tx{};
652 const std::size_t tx_len = modbus::BuildWriteSingle(address_, reg_addr, value,
653 tx.data(), tx.size());
654 if (tx_len == 0) return DriverResult<void>::failure(DriverError::BufferTooSmall);
655
656 DriverError rc = SendRequest(tx.data(), tx_len);
657 if (rc != DriverError::None) {
658 return DriverResult<void>{LogFailure(rc, reg::fn::WriteSingle, reg_addr)};
659 }
660
661 // Broadcast → no response.
662 if (address_ == kBroadcastAddress) return DriverResult<void>::success();
663
664 std::array<uint8_t, 8> rx{};
665 std::size_t got = 0;
666 rc = ReceiveReply(reg::fn::WriteSingle, rx.data(), rx.size(), got);
667 if (rc == DriverError::None) {
668 rc = modbus::ParseWriteSingleEcho(rx.data(), got, tx.data(), tx_len);
669 }
670 return DriverResult<void>{LogFailure(rc, reg::fn::WriteSingle, reg_addr)};
671 }

679 DriverResult<void> WriteMultipleRegisters(uint16_t reg_addr,
680 const uint16_t* regs, uint8_t count) noexcept {
681 if (!regs || count == 0 || count > 32) {
682 return DriverResult<void>::failure(DriverError::InvalidParameter);
683 }
684 std::array<uint8_t, 9 + 32 * 2> tx{};
685 const std::size_t tx_len = modbus::BuildWriteMultiple(address_, reg_addr, regs, count,
686 tx.data(), tx.size());
687 if (tx_len == 0) return DriverResult<void>::failure(DriverError::BufferTooSmall);
688
689 DriverError rc = SendRequest(tx.data(), tx_len);
690 if (rc != DriverError::None) {
691 return DriverResult<void>{LogFailure(rc, reg::fn::WriteMultiple, reg_addr)};
692 }

695

698 rc = ReceiveReply(reg::fn::WriteMultiple, rx.data(), rx.size(), got);

720 DriverError LogFailure(DriverError rc, uint8_t fn, uint16_t reg_addr) noexcept {
721 if (rc == DriverError::None || quiet_) {
722 return rc;
723 }
724 if (rc == DriverError::ModbusException) {
725 uart_.log(kLogTag, "addr %u fn 0x%02X reg %u: ModbusException %s",
726 static_cast<unsigned>(address_), static_cast<unsigned>(fn),
727 static_cast<unsigned>(reg_addr), ToString(last_exception_).data());
728 } else {
729 uart_.log(kLogTag, "addr %u fn 0x%02X reg %u: %s", static_cast<unsigned>(address_),
730 static_cast<unsigned>(fn), static_cast<unsigned>(reg_addr), ToString(rc).data());
731 }
732 return rc;
733 }

747 DriverError SendRequest(const uint8_t* tx, std::size_t tx_len) noexcept {
748 last_exception_ = ModbusExceptionCode::None;
749 uart_.delay_ms(modbus::InterFrameGapMs(bus_bps_));
750 uart_.flush_rx();
751 return uart_.write(tx, tx_len) ? DriverError::None : DriverError::SerialError;
752 }

769 DriverError ReceiveReply(uint8_t fn, uint8_t* rx, std::size_t expected_len,
770 std::size_t& got) noexcept {
771 got = uart_.read(rx, modbus::kReplyHeaderLength, timeout_ms_);
772 if (got == 0)
773 return DriverError::Timeout;
774 if (got < modbus::kReplyHeaderLength)
775 return DriverError::InvalidFrame;
776 if (modbus::IsExceptionReply(rx, got)) {
777 got += uart_.read(rx + got, modbus::kExceptionFrameLength - got, timeout_ms_);
778 uint8_t code = 0;
779 const DriverError rc = modbus::ParseException(rx, got, address_, fn, &code);
780 if (rc == DriverError::ModbusException) {
781 last_exception_ = static_cast<ModbusExceptionCode>(code);
782 }
783 return rc;
784 }
785 got += uart_.read(rx + got, expected_len - got, timeout_ms_);
786 return DriverError::None;
787 }
