36inline uint16_t Crc16(const uint8_t* data, std::size_t length) noexcept {
37 uint16_t crc = 0xFFFF;
38 for (std::size_t i = 0; i < length; ++i) {
39 crc ^= static_cast<uint16_t>(data[i]);
40 for (uint8_t b = 0; b < 8; ++b) {
41 if (crc & 0x0001U) {
42 crc = static_cast<uint16_t>((crc >> 1) ^ 0xA001U);
43 } else {
44 crc >>= 1;
45 }
46 }
47 }
48 return crc;
49}

52inline std::size_t AppendCrc(uint8_t* frame, std::size_t length) noexcept {
53 const uint16_t crc = Crc16(frame, length);
54 frame[length] = static_cast<uint8_t>(crc & 0x00FFU);
55 frame[length + 1] = static_cast<uint8_t>((crc >> 8) & 0x00FFU);
56 return length + 2;
57}

60inline bool CheckCrc(const uint8_t* frame, std::size_t length) noexcept {
61 if (length < 4) return false;
62 const uint16_t want = Crc16(frame, length - 2);
63 const uint16_t got = static_cast<uint16_t>(frame[length - 2]) |
64 (static_cast<uint16_t>(frame[length - 1]) << 8);
65 return want == got;
66}

74inline std::size_t BuildReadHolding(uint8_t unit_addr,
75 uint16_t reg_addr,
76 uint16_t reg_count,
77 uint8_t* out, std::size_t out_capacity) noexcept {
78 if (out_capacity < 8) return 0;
79 out[0] = unit_addr;
80 out[1] = reg::fn::ReadHolding;
81 out[2] = static_cast<uint8_t>((reg_addr >> 8) & 0xFFU);

83 out[4] = static_cast<uint8_t>((reg_count >> 8) & 0xFFU);
84 out[5] = static_cast<uint8_t>(reg_count & 0xFFU);
85 return AppendCrc(out, 6);

105inline std::size_t BuildWriteMultiple(uint8_t unit_addr,
106 uint16_t reg_addr,
107 const uint16_t* regs, uint8_t count,
108 uint8_t* out, std::size_t out_capacity) noexcept {
109 const std::size_t needed = 9 + count * 2;
110 if (count == 0 || out_capacity < needed) return 0;
111 out[0] = unit_addr;
112 out[1] = reg::fn::WriteMultiple;
113 out[2] = static_cast<uint8_t>((reg_addr >> 8) & 0xFFU);
114 out[3] = static_cast<uint8_t>(reg_addr & 0xFFU);
115 out[4] = static_cast<uint8_t>((count >> 8) & 0xFFU);
116 out[5] = static_cast<uint8_t>(count & 0xFFU);
117 out[6] = static_cast<uint8_t>(count * 2);
118 for (uint8_t i = 0; i < count; ++i) {
119 out[7 + i * 2] = static_cast<uint8_t>((regs[i] >> 8) & 0xFFU);
120 out[7 + i * 2 + 1] = static_cast<uint8_t>(regs[i] & 0xFFU);
121 }
122 return AppendCrc(out, 7 + count * 2);
123}

145constexpr uint32_t InterFrameGapUs(uint32_t bps) noexcept {
146 if (bps == 0)
147 bps = 4800;
148 if (bps > kFixedGapAboveBps)
149 return kFixedGapUs;
150 // 3.5 * 11 bits * 1e6 us / bps, rounded up.
151 constexpr uint32_t kNumerator = 7U * kRtuBitsPerChar * 1000000U / 2U; // 38'500'000
152 return (kNumerator + bps - 1U) / bps;
153}

183inline DriverError ParseException(const uint8_t* frame, std::size_t length, uint8_t expected_addr,
184 uint8_t expected_fn, uint8_t* code) noexcept {
185 if (length != kExceptionFrameLength)
186 return DriverError::InvalidFrame;
187 if (!CheckCrc(frame, length))
188 return DriverError::BadCrc;
189 if (frame[0] != expected_addr)
190 return DriverError::ResponseMismatch;
191 if (frame[1] != (expected_fn | reg::fn::ExceptionMask))
192 return DriverError::ResponseMismatch;
193 if (code)
194 *code = frame[2];
195 return DriverError::ModbusException;
196}

200inline DriverError ParseReadHolding(const uint8_t* frame, std::size_t length,
201 uint8_t expected_addr, uint8_t expected_count,
202 uint16_t* regs) noexcept {
203 if (length < 5) return DriverError::InvalidFrame;
204 if (!CheckCrc(frame, length)) return DriverError::BadCrc;
205 if (frame[0] != expected_addr) return DriverError::ResponseMismatch;
206 if (frame[1] & reg::fn::ExceptionMask) return DriverError::ModbusException;
207 if (frame[1] != reg::fn::ReadHolding) return DriverError::ResponseMismatch;
208 const std::size_t byte_count = frame[2];
209 if (byte_count != expected_count * 2U) return DriverError::InvalidFrame;
210 if (length < 3 + byte_count + 2) return DriverError::InvalidFrame;
211 for (uint8_t i = 0; i < expected_count; ++i) {
212 regs[i] = (static_cast<uint16_t>(frame[3 + i * 2]) << 8) |
213 static_cast<uint16_t>(frame[3 + i * 2 + 1]);
214 }
215 return DriverError::None;
216}

220inline DriverError ParseWriteSingleEcho(const uint8_t* frame, std::size_t length,
221 const uint8_t* request, std::size_t req_len) noexcept {
222 if (length < req_len) return DriverError::InvalidFrame;
223 if (!CheckCrc(frame, length)) return DriverError::BadCrc;
224 if (frame[0] != request[0]) return DriverError::ResponseMismatch;
225 if (frame[1] & reg::fn::ExceptionMask) return DriverError::ModbusException;
226 for (std::size_t i = 0; i < req_len; ++i) {
227 if (frame[i] != request[i]) return DriverError::ResponseMismatch;
228 }
229 return DriverError::None;
230}

233inline DriverError ParseWriteMultipleEcho(const uint8_t* frame, std::size_t length,
234 uint8_t expected_addr,
235 uint16_t expected_reg_addr,
236 uint16_t expected_count) noexcept {
237 if (length < 8) return DriverError::InvalidFrame;
238 if (!CheckCrc(frame, length)) return DriverError::BadCrc;
239 if (frame[0] != expected_addr) return DriverError::ResponseMismatch;
240 if (frame[1] & reg::fn::ExceptionMask) return DriverError::ModbusException;
241 if (frame[1] != reg::fn::WriteMultiple) return DriverError::ResponseMismatch;
242 const uint16_t got_addr = (static_cast<uint16_t>(frame[2]) << 8) | frame[3];
243 const uint16_t got_cnt = (static_cast<uint16_t>(frame[4]) << 8) | frame[5];
244 if (got_addr != expected_reg_addr) return DriverError::ResponseMismatch;
245 if (got_cnt != expected_count) return DriverError::ResponseMismatch;
246 return DriverError::None;
247}
